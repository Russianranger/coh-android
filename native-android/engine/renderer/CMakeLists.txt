cmake_minimum_required(VERSION 3.18)
project(coh_renderer C)
add_library(coh_renderer STATIC coh_renderer.c)
target_compile_features(coh_renderer PUBLIC c_std_99)
target_include_directories(coh_renderer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(coh_renderer PUBLIC m)
option(COH_RENDERER_GLES "Build the GLES3 driver adapter" ${ANDROID})
if(COH_RENDERER_GLES)
  target_sources(coh_renderer PRIVATE coh_renderer_gles.c)
  target_link_libraries(coh_renderer PUBLIC GLESv3)
endif()
