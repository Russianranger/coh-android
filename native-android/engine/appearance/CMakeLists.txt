cmake_minimum_required(VERSION 3.22.1)
# Opt-in only; accepted APKs do not include this directory.
add_library(coh_appearance STATIC coh_appearance.c)
target_compile_features(coh_appearance PUBLIC c_std_11)
target_include_directories(coh_appearance PUBLIC .)
target_compile_options(coh_appearance PRIVATE -Wall -Wextra -Werror -pedantic)
option(COH_APPEARANCE_WITH_ADAPTERS "Link existing GEO/rig/palette APIs" OFF)
if(COH_APPEARANCE_WITH_ADAPTERS)
    if(NOT TARGET coh_engine_assets OR NOT TARGET coh_appearance_character_kernel)
        message(FATAL_ERROR "Provide coh_engine_assets and coh_appearance_character_kernel (unchanged character/coh_character.c) first; do not compile animation.c twice")
    endif()
    target_sources(coh_appearance PRIVATE coh_appearance_adapters.c)
    target_link_libraries(coh_appearance PUBLIC coh_engine_assets coh_appearance_character_kernel)
endif()
option(COH_APPEARANCE_WITH_RENDER_BRIDGE "Add native GEO conversion and CPU skin render adapter" OFF)
if(COH_APPEARANCE_WITH_RENDER_BRIDGE)
    if(NOT COH_APPEARANCE_WITH_ADAPTERS OR NOT TARGET coh_appearance_mesh_kernel)
        message(FATAL_ERROR "Enable appearance adapters and provide coh_appearance_mesh_kernel (unchanged coh_mesh.c) once")
    endif()
    target_sources(coh_appearance PRIVATE coh_appearance_mesh.c coh_appearance_render.c coh_appearance_actor.c)
    target_link_libraries(coh_appearance PUBLIC coh_appearance_mesh_kernel)
endif()
