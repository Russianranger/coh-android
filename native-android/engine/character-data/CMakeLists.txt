cmake_minimum_required(VERSION 3.16)
project(coh_native_character_data C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT TARGET coh_native_replication)
  add_subdirectory(../replication ${CMAKE_CURRENT_BINARY_DIR}/replication)
endif()
add_library(coh_native_character_data STATIC coh_character_data.c)
target_include_directories(coh_native_character_data PUBLIC .)
target_compile_options(coh_native_character_data PRIVATE -Wall -Wextra -Werror)
target_link_libraries(coh_native_character_data PUBLIC coh_native_replication m)
if(COH_CHARACTER_DATA_FIXTURE_DIR)
  add_executable(coh_character_data_test test_character_data.c)
  target_include_directories(coh_character_data_test PRIVATE ${COH_CHARACTER_DATA_FIXTURE_DIR})
  target_compile_options(coh_character_data_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(coh_character_data_test PRIVATE coh_native_character_data)
endif()
