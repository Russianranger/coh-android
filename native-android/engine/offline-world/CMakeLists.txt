cmake_minimum_required(VERSION 3.22.1)
if(NOT TARGET coh_engine_assets)
    message(FATAL_ERROR "Add engine/assets before engine/offline-world")
endif()
add_library(coh_engine_offline_world STATIC coh_offline_world.c)
target_include_directories(coh_engine_offline_world PUBLIC .)
target_compile_features(coh_engine_offline_world PUBLIC c_std_11)
target_compile_options(coh_engine_offline_world PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
target_link_libraries(coh_engine_offline_world PUBLIC coh_engine_assets m)
