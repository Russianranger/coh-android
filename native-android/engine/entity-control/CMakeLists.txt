cmake_minimum_required(VERSION 3.16)
project(coh_native_entity_control C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT TARGET coh_native_replication)
  add_subdirectory(../replication ${CMAKE_CURRENT_BINARY_DIR}/replication)
endif()
add_library(coh_native_entity_control STATIC coh_entity_control.c)
target_include_directories(coh_native_entity_control PUBLIC .)
target_compile_options(coh_native_entity_control PRIVATE -Wall -Wextra -Werror)
target_link_libraries(coh_native_entity_control PUBLIC coh_native_replication m)
if(COH_ENTITY_CONTROL_FIXTURE_DIR)
  add_executable(coh_entity_control_test test_entity_control.c)
  target_include_directories(coh_entity_control_test PRIVATE ${COH_ENTITY_CONTROL_FIXTURE_DIR})
  target_compile_options(coh_entity_control_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(coh_entity_control_test PRIVATE coh_native_entity_control)
endif()
