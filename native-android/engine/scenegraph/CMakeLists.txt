cmake_minimum_required(VERSION 3.16)
project(coh_native_scenegraph C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(coh_native_scenegraph STATIC coh_scenegraph.c coh_scene_metadata.c coh_scene_lookup.c ../replication/coh_replication.c ../replication/coh_groups.c)
target_include_directories(coh_native_scenegraph PUBLIC .)
target_compile_options(coh_native_scenegraph PRIVATE -Wall -Wextra -Werror)
target_link_libraries(coh_native_scenegraph PUBLIC m)
option(COH_SCENEGRAPH_TESTS "Build standalone graph oracle tests" OFF)
if(COH_SCENEGRAPH_TESTS)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  get_filename_component(COH_SCENEGRAPH_DEFAULT_SOURCE "../../.." ABSOLUTE)
  set(COH_SCENEGRAPH_SOURCE_ROOT "${COH_SCENEGRAPH_DEFAULT_SOURCE}" CACHE PATH "Repo containing pinned original source")
  set(COH_SCENEGRAPH_FIXTURE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check.py" --source-root "${COH_SCENEGRAPH_SOURCE_ROOT}" --prepare-only --fixtures-output "${COH_SCENEGRAPH_FIXTURE_DIR}" RESULT_VARIABLE fixture_status)
  if(NOT fixture_status EQUAL 0)
    message(FATAL_ERROR "Pinned original graph fixtures could not be prepared")
  endif()
  add_executable(coh_scenegraph_test tests/test_scenegraph.c tests/test_scene_lookup.c ../scenecatalog/coh_scenecatalog.c)
  target_include_directories(coh_scenegraph_test PRIVATE "${COH_SCENEGRAPH_FIXTURE_DIR}")
  target_compile_options(coh_scenegraph_test PRIVATE -Wall -Wextra -Werror)
  target_link_options(coh_scenegraph_test PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=free)
  target_link_libraries(coh_scenegraph_test PRIVATE coh_native_scenegraph)
  enable_testing()
  add_test(NAME scenegraph COMMAND coh_scenegraph_test)
endif()
