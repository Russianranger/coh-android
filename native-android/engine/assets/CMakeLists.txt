cmake_minimum_required(VERSION 3.22.1)
# Opt-in only. Nothing includes this from existing APK builds.
add_library(coh_engine_asset_kernels STATIC ../../src/coh_assets.c ../../src/coh_dds.c)
target_include_directories(coh_engine_asset_kernels PUBLIC ../../include)
add_library(coh_engine_assets STATIC coh_asset_loader.c coh_asset_geo.c)
target_include_directories(coh_engine_assets PUBLIC .)
target_compile_features(coh_engine_assets PUBLIC c_std_99)
target_compile_options(coh_engine_assets PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
if(ANDROID)
    target_link_libraries(coh_engine_assets PUBLIC coh_engine_asset_kernels z m)
else()
    find_package(ZLIB REQUIRED)
    target_link_libraries(coh_engine_assets PUBLIC coh_engine_asset_kernels ZLIB::ZLIB m)
endif()
option(COH_ASSET_WITH_AUDIO "Add adapter to existing coh_native_decode target" OFF)
if(COH_ASSET_WITH_AUDIO)
    if(NOT TARGET coh_native_decode)
        message(FATAL_ERROR "Add native-android/decode first; its pinned dependencies are mandatory")
    endif()
    target_sources(coh_engine_assets PRIVATE coh_asset_audio.c)
    target_link_libraries(coh_engine_assets PUBLIC coh_native_decode)
endif()
