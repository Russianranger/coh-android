cmake_minimum_required(VERSION 3.16)
project(coh_native_scenecatalog C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_subdirectory(../scenegraph scenegraph)
add_library(coh_native_scenecatalog STATIC coh_scenecatalog.c)
target_include_directories(coh_native_scenecatalog PUBLIC .)
target_compile_options(coh_native_scenecatalog PRIVATE -Wall -Wextra -Werror)
target_link_libraries(coh_native_scenecatalog PUBLIC coh_native_scenegraph)
option(COH_SCENECATALOG_TESTS "Build standalone original-source catalog tests" OFF)
if(COH_SCENECATALOG_TESTS)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  get_filename_component(COH_SCENECATALOG_DEFAULT_SOURCE "../../.." ABSOLUTE)
  set(COH_SCENECATALOG_SOURCE_ROOT "${COH_SCENECATALOG_DEFAULT_SOURCE}" CACHE PATH "Repo containing pinned original source")
  set(COH_SCENECATALOG_FIXTURE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check.py" --source-root "${COH_SCENECATALOG_SOURCE_ROOT}" --prepare-only --fixtures-output "${COH_SCENECATALOG_FIXTURE_DIR}" RESULT_VARIABLE fixture_status)
  if(NOT fixture_status EQUAL 0)
    message(FATAL_ERROR "Pinned original catalog fixtures could not be prepared")
  endif()
  add_executable(coh_scenecatalog_test tests/test_scenecatalog.c)
  target_include_directories(coh_scenecatalog_test PRIVATE "${COH_SCENECATALOG_FIXTURE_DIR}")
  target_compile_options(coh_scenecatalog_test PRIVATE -Wall -Wextra -Werror)
  target_link_options(coh_scenecatalog_test PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=free)
  target_link_libraries(coh_scenecatalog_test PRIVATE coh_native_scenecatalog)
  enable_testing()
  add_test(NAME scenecatalog COMMAND coh_scenecatalog_test)
endif()
