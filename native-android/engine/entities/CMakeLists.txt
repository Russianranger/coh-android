cmake_minimum_required(VERSION 3.16)
project(coh_native_entities C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT TARGET coh_native_replication)
  add_subdirectory(../replication ${CMAKE_CURRENT_BINARY_DIR}/replication)
endif()
add_library(coh_native_entities STATIC coh_entities.c coh_costume.c)
target_include_directories(coh_native_entities PUBLIC .)
target_compile_options(coh_native_entities PRIVATE -Wall -Wextra -Werror)
target_link_libraries(coh_native_entities PUBLIC coh_native_replication m)
# Fixture generation executes pinned original functions on the host. Never try
# to run the historical x86 oracle during an Android cross compilation.
if(COH_ENTITIES_FIXTURE_DIR)
  add_executable(coh_entities_test test_entities.c test_costume.c)
  target_include_directories(coh_entities_test PRIVATE ${COH_ENTITIES_FIXTURE_DIR})
  target_compile_options(coh_entities_test PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(coh_entities_test PRIVATE coh_native_entities)
endif()
