cmake_minimum_required(VERSION 3.22.1)
# Run tools/decode_fixture/acquire.py before configuring. Codec sources are
# byte-pinned and compiled statically; there is no app-time network dependency.
set(COH_CODEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/.deps")
if(NOT EXISTS "${COH_CODEC_ROOT}/ogg/include/ogg/config_types.h")
    message(FATAL_ERROR "Run python3 native-android/tools/decode_fixture/acquire.py first")
endif()
set(COH_VORBIS_FILES mdct smallft block envelope window lsp lpc analysis synthesis
    psy info floor1 floor0 res0 mapping0 registry codebook sharedbook lookup bitrate vorbisfile)
set(COH_VORBIS_SOURCES "")
foreach(name IN LISTS COH_VORBIS_FILES)
    if(name STREQUAL "sharedbook")
        list(APPEND COH_VORBIS_SOURCES "${COH_CODEC_ROOT}/safe/sharedbook.c")
    else()
        list(APPEND COH_VORBIS_SOURCES "${COH_CODEC_ROOT}/vorbis/lib/${name}.c")
    endif()
endforeach()
add_library(coh_pinned_ogg STATIC "${COH_CODEC_ROOT}/ogg/src/bitwise.c" "${COH_CODEC_ROOT}/ogg/src/framing.c")
add_library(coh_pinned_vorbis STATIC ${COH_VORBIS_SOURCES})
set_target_properties(coh_pinned_ogg coh_pinned_vorbis PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(coh_pinned_ogg PUBLIC "${COH_CODEC_ROOT}/ogg/include")
target_include_directories(coh_pinned_vorbis PUBLIC "${COH_CODEC_ROOT}/vorbis/include"
    PRIVATE "${COH_CODEC_ROOT}/vorbis/lib")
target_link_libraries(coh_pinned_vorbis PUBLIC coh_pinned_ogg m)
target_compile_options(coh_pinned_vorbis PRIVATE -ffp-contract=off)
target_compile_definitions(coh_pinned_vorbis PRIVATE HAVE_ALLOCA_H=1)
add_library(coh_native_decode STATIC coh_decode.c)
set_target_properties(coh_native_decode PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(coh_native_decode PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_features(coh_native_decode PRIVATE c_std_11)
target_compile_options(coh_native_decode PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
target_link_libraries(coh_native_decode PUBLIC coh_pinned_vorbis)
