cmake_minimum_required(VERSION 3.16)
project(coh_postgresql_probe C)
find_package(ODBC REQUIRED)
add_executable(odbc_probe odbc_probe.c)
target_include_directories(odbc_probe PRIVATE ../overlay/Common/sql ${ODBC_INCLUDE_DIRS})
target_link_libraries(odbc_probe PRIVATE ${ODBC_LIBRARIES})
set_property(TARGET odbc_probe PROPERTY C_STANDARD 99)
if(MSVC)
  target_compile_definitions(odbc_probe PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(odbc_probe PRIVATE -Wall -Wextra -Werror)
endif()
